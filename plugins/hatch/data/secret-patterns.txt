# secret-patterns.txt — the single source for the secret guard's regexes.
# Three named patterns, one per line as NAME=<extended regex>. hooks/pre-commit reads this file;
# templates/global-guard/_env-guard.sh carries the same three values because a global hook cannot
# depend on a plugin path — scripts/check.sh asserts the two stay identical.
SAFE_ENV_PATTERN=(^|/)\.env(\.[A-Za-z0-9_-]+)*\.(example|sample|template)$
BLOCKED_PATTERN=(^|/)\.env($|\.[^/]*$)|(^|/)\.dev\.vars($|\.[^/]*$)|\.pem$|\.p12$|\.pfx$|(^|/)id_rsa$|(^|/)id_ed25519$|(^|/)id_ecdsa$|(^|/)\.npmrc$|(^|/)\.pypirc$|(^|/)\.netrc$|(^|/)credentials\.json$|(^|/)service-account.*\.json$|\.tfvars$
